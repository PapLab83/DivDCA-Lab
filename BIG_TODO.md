## Проект: Платформа для бэктестинга LLM-инвестиционных стратегий

### Позиционирование
> Production-ready multi-agent система для бэктестинга инвестиционных стратегий на базе LLM с жесткой изоляцией от look-ahead bias, спроектированная для воспроизводимости, расширяемости и наблюдаемости.

CREATE TABLE ticker_mapping (

    id SERIAL PRIMARY KEY,

    momentum_12m DECIMAL(8,4),
    volatility_60d DECIMAL(8,4),
    -- другие метрики
    CONSTRAINT fk_company FOREIGN KEY (company_id) REFERENCES ticker_mapping(company_id)
);

-- Таблица решений (ядро системы)
CREATE TABLE portfolio_actions (
    id SERIAL PRIMARY KEY,
    backtest_id UUID NOT NULL,               -- идентификатор запуска
    date DATE NOT NULL,                      -- дата действия
    action VARCHAR(10) NOT NULL,             -- buy, sell, dividend, split
    company_id INT NOT NULL,
    shares INT,
    price DECIMAL(12,4),
    cash_flow DECIMAL(12,2),
    rationale TEXT,                          -- объяснение от агента
    agent_version VARCHAR(20),               -- версия промпта/агента
    prompt_hash VARCHAR(64),                 -- для воспроизводимости
    created_at TIMESTAMP DEFAULT NOW()
);

-- Результаты бэктеста (материализованный view или таблица)
CREATE TABLE backtest_results (
    id SERIAL PRIMARY KEY,
    backtest_id UUID NOT NULL,
    date DATE NOT NULL,
    portfolio_value DECIMAL(16,2),
    cash DECIMAL(16,2),
    return_ytd DECIMAL(10,4),
    benchmark_value DECIMAL(16,2)
);

-- Кэш LLM ответов (для экономии)
CREATE TABLE llm_cache (
    id SERIAL PRIMARY KEY,
    prompt_hash VARCHAR(64) UNIQUE,
    prompt TEXT,
    response TEXT,
    model VARCHAR(20),
    tokens_used INT,
    created_at TIMESTAMP DEFAULT NOW()
);

## 4. Ключевые технические решения (для Senior/Lead уровня)

### 4.1. Защита от look-ahead bias

```python
class SafeSQLExecutor:
    def __init__(self, db_connection, current_date):
        self.db = db_connection
        self.current_date = current_date
    
    def execute(self, sql_query):
        # 1. Парсим запрос, находим WHERE условия с date
        # 2. Если есть date > {something} - заменяем на date <= current_date
        # 3. Принудительно добавляем AND date <= current_date если нет
        # 4. Добавляем LIMIT если нет
        # 5. Логируем исходный и измененный SQL
        safe_sql = self._inject_time_filter(sql_query)
        return self.db.execute(safe_sql)
```

### 4.2. Агентная архитектура (без LangChain для контроля)

```python
class Agent:
    def __init__(self, name, system_prompt, tools, llm_client):
        self.name = name
        self.system_prompt = system_prompt
        self.tools = tools
        self.llm = llm_client
        self.conversation_history = []
    
    def call(self, user_input, context=None):
        # Формируем промпт с current_date
        full_prompt = f"{self.system_prompt}\nТекущая дата: {context['current_date']}\n{user_input}"
        
        # Вызываем LLM с tools
        response = self.llm.chat(
            messages=[{"role": "system", "content": full_prompt}] + self.conversation_history,
            tools=self.tools,
            tool_choice="auto"
        )
        
        # Логируем токены, стоимость, промпт
        self._log_call(full_prompt, response)
        
        return response
```
