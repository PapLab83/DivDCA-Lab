Верни JSON в следующем формате:
{
  "events": [
    {
      "title": "Краткое название (2-5 слов)",
      "description": "Подробное описание события и его влияния на компанию или цену акций",
      "impact": "positive/negative/neutral",  // как это повлияло на компанию
      "price_impact": "positive/negative/neutral",  // как это повлияло на цену акций (может отличаться от impact)
      "confidence": 85  // число от 0 до 100
    }
  ]
}
